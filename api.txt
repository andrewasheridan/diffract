package sheridan.diffract
├── __version__: str
├── class ApiDiff
│   ├── added: tuple[NameChange, ...]
│   ├── removed: tuple[NameChange, ...]
│   ├── changed: tuple[SignatureChange, ...]
│   ├── property breaking_changes: tuple[SignatureChange, ...]
│   ├── property additive_changes: tuple[SignatureChange, ...]
│   ├── property is_breaking: bool
│   ├── property has_additions: bool
│   └── property is_empty: bool
├── class ChangeKind(StrEnum)
│   ├── added
│   └── removed
├── class CommitType(StrEnum)
│   ├── feat_breaking
│   ├── feat
│   ├── code
│   ├── test
│   ├── other
│   └── property possible_types: tuple[str, ...]
├── class DiffractConfig
│   ├── src: str | None
│   └── tests: str | None
├── class DiffractError(Exception)
├── class DiffractResult
│   ├── commit_type: CommitType
│   ├── summary: str
│   ├── diff: ApiDiff
│   ├── base_ref: str
│   ├── head_ref: str
│   └── def to_dict(self) -> dict[str, object]
├── class GitError(DiffractError)
├── class NameChange
│   ├── module: str
│   ├── name: str
│   └── kind: ChangeKind
├── class SignatureChange
│   ├── module: str
│   ├── name: str
│   ├── detail: str
│   └── is_breaking: bool
├── class SurfaceError(DiffractError)
├── def check(base_ref: str = 'HEAD~1', head_ref: str = 'HEAD', repo_path: Path | None = None, src_path: str = 'src', tests_path: str = 'tests') -> DiffractResult
├── def check_staged(repo_path: Path | None = None, src_path: str = 'src', tests_path: str = 'tests') -> DiffractResult
└── def load_config(repo_root: Path) -> DiffractConfig
